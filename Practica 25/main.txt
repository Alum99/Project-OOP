# Глобальные переменные
soldiers = []      # общий список всех солдат
units = {}         # подразделения: имя подразделения - список солдат


# Функции

def create_soldier(name): # Создание солдата
    soldier = {"name": name} # Создаётся словарь-солдат
    soldiers.append(soldier) # общий список всех солдат
    return soldier


def create_unit(name):   # Создание подразделения (только список солдат, без вложений)
    units[name] = []     # каждое подразделение содержит только солдат
    return name          # Возвращается строка — имя подразделения


def add_soldier_to_unit(unit_name, soldier): # Добавляет солдата в подразделение
    if unit_name in units: # существует ли подразделение
        units[unit_name].append(soldier) # добавляем солдата в его список
    else:
        print(f"Ошибка: подразделения '{unit_name}' не существует")


def remove_soldier_from_unit(unit_name, soldier):          # Удаляет солдата из подразделения
    if unit_name in units and soldier in units[unit_name]: # существует ли подразделение и есть ли солдат внутри него
        units[unit_name].remove(soldier)                   # Удаляем солдата
    else:
        print("Ошибка: убрать солдата невозможно")


def march_soldier(soldier): #  Команда марш одному солдату
    print(f"{soldier['name']} марширует!")


def march_unit(unit_name): #  Команда марш всем солдатам данного подразделения
    if unit_name not in units:
        print(f"Подразделение '{unit_name}' не найдено!")
        return

    print(f"Подразделение '{unit_name}' начинает марш:")
    for soldier in units[unit_name]:
        march_soldier(soldier)
    print(f"Подразделение '{unit_name}' закончило марш.\n")


# Демонстрация
if __name__ == "__main__":

    # создаём солдат
    s1 = create_soldier("Дмитрий")
    s2 = create_soldier("Пётр")
    s3 = create_soldier("Владимир")
    s4 = create_soldier("Михаил")

    # создаём подразделения
    platoon1 = create_unit("Взвод 1")
    platoon2 = create_unit("Взвод 2")

    # добавляем солдат
    add_soldier_to_unit(platoon1, s1)
    add_soldier_to_unit(platoon1, s2)

    add_soldier_to_unit(platoon2, s3)
    add_soldier_to_unit(platoon2, s4)

    # команды
    print("Команда одному солдату:")
    march_soldier(s1)

    print("\nКоманда взводу 1:")
    march_unit(platoon1)

    print("Команда взводу 2:")
    march_unit(platoon2)
