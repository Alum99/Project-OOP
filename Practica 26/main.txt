from abc import ABC, abstractmethod    # ABC - абстрактный базовый класс

class Component(ABC):                  # абстрактный класс, наследуется от ABC
    @abstractmethod
    def operation(self):               # абстрактный метод operation()
        pass
    
    @abstractmethod
    def add(self, component):          # абстрактный метод add(component)
        pass
    
    @abstractmethod
    def remove(self, component):       # абстрактный метод remove(component)
        pass
    
    @abstractmethod
    def march(self):                   # абстрактный метод march()
        pass


class Soldier(Component):      # класс, наследуется от Component
    def __init__(self, name):
        self.name = name       # name сохраняет в атрибут экземпляра
    
    def operation(self):
        pass
    
    def add(self, member):
        pass
    
    def remove(self, member):
        pass
    
    def march(self):
        pass


class Unit(Component):         # класс, наследуется от Component
    def __init__(self, name):
        self.name = name       # name сохраняет в атрибут экземпляра
        self.members = []      # инициализирует атрибут self.members пустым списком
    
    def operation(self):
        pass
    
    def add(self, member):
        pass
    
    def remove(self, member):
        pass
    
    def march(self):
        pass
